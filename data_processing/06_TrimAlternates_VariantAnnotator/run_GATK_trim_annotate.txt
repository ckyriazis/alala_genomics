GATK_trim_annotate() {

CHROM=$1

INDIR=/home/centos/USS/chris/alala/output/data_processing/05_GenotypeGVCFs
OUTDIR=/home/centos/USS/chris/alala/output/data_processing/06_TrimAlternates_VariantAnnotator
TEMPDIR=/home/centos/USS/chris/alala/scripts/data_processing/06_TrimAlternates_VariantAnnotator/tmp
REFERENCE=/home/centos/USS/chris/alala/reference_genome/GCF_020740725.1_bCorHaw1.pri.cur_genomic.fna
GATK=~/USS/conservation-genetics/nwr/bin/GenomeAnalysisTK.jar
BED=/home/centos/USS/chris/alala/reference_genome/scaffold_files/scaffold_${CHROM}.bed


LOG=/home/centos/USS/chris/alala/scripts/data_processing/06_TrimAlternates_VariantAnnotator/logfiles/TrimAlternates_chrom${CHROM}_log.txt

date > ${LOG}

wait

java -jar -Djava.io.tmpdir=${TEMPDIR} ${GATK} \
-T SelectVariants \
-R ${REFERENCE} \
-trimAlternates \
-L ${BED} \
-V ${INDIR}/175AL_noFilter_scaff${CHROM}.vcf.gz \
-o ${OUTDIR}/175AL_noFilter_TrimAlternates_chrom${CHROM}.vcf.gz &>> ${LOG}

date >> ${LOG}


LOG=/home/centos/USS/chris/alala/scripts/data_processing/06_TrimAlternates_VariantAnnotator/logfiles/VariantAnnotator_chrom${CHROM}_log.txt

date > ${LOG}

java -jar -Djava.io.tmpdir=${TEMPDIR} ${GATK} \
-T VariantAnnotator \
-R ${REFERENCE} \
-G StandardAnnotation \
-A VariantType \
-A AlleleBalance \
-L ${BED} \
-V ${OUTDIR}/175AL_noFilter_TrimAlternates_chrom${CHROM}.vcf.gz \
-o ${OUTDIR}/175AL_noFilter_VariantAnnotator_chrom${CHROM}.vcf.gz &>> ${LOG}

date >> ${LOG}

}

#export the function
export -f GATK_trim_annotate

seq 43 | parallel -j 12 GATK_trim_annotate
